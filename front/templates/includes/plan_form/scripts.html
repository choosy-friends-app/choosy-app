(function () { const form = document.querySelector("[data-plan-form]"); if (!form) return; const locationInput = form.querySelector("[data-location-input]"); const addressInput = form.querySelector("[data-address-input]"); } function searchLocations(query, options) { const trimmed = query.trim(); if (trimmed.length < 3) { hideSuggestions(); setLocationStatus("Type at least 3 characters to search OpenStreetMap.", ""); return Promise.resolve([]);
